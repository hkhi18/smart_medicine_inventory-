// Signed session cookie (HMAC-SHA256 with Web Crypto, so it also works inside proxy.js).
// The cookie holds only the user name and an expiry time; everything else is read from the database.
__turbopack_context__.s([
    "COOKIE_NAME",

const COOKIE_NAME = 'smi_session';
const MAX_AGE_SECONDS = 8 * 60 * 60; // 8 hours
const enc = new TextEncoder();
function secret() {
    if (process.env.SESSION_SECRET) return process.env.SESSION_SECRET;
    if ("TURBOPACK compile-time falsy", 0) //TURBOPACK unreachable
    ;
    return 'dev-only-secret-change-me-in-env-local';
}
function toB64url(bytes) {
    let s = '';
    bytes.forEach((b)=>{
        s += String.fromCharCode(b);
    });
    return btoa(s).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
}
function fromB64url(str) {
    let s = str.replace(/-/g, '+').replace(/_/g, '/');
    while(s.length % 4)s += '=';
    return atob(s);
}
async function hmac(data) {
    const key = await crypto.subtle.importKey('raw', enc.encode(secret()), {
        name: 'HMAC',
        hash: 'SHA-256'
    }, false, [
        'sign'
    ]);
    const sig = await crypto.subtle.sign('HMAC', key, enc.encode(data));
    return toB64url(new Uint8Array(sig));
}
// Compare two strings without stopping at the first different character
function safeEqual(a, b) {
    if (a.length !== b.length) return false;
    let diff = 0;
    for(let i = 0; i < a.length; i++)diff |= a.charCodeAt(i) ^ b.charCodeAt(i);
    return diff === 0;
}
async function signSession(username) {
    const payload = toB64url(enc.encode(JSON.stringify({
        u: username,
        exp: Date.now() + MAX_AGE_SECONDS * 1000
    })));
    return `${payload}.${await hmac(payload)}`;
}
async function verifySession(token) {
    if (!token || !token.includes('.')) return null;
    const [payload, sig] = token.split('.');
    if (!payload || !sig || !safeEqual(sig, await hmac(payload))) return null;
    try {
        const data = JSON.parse(fromB64url(payload));
        return data.exp > Date.now() && data.u ? {
            u: data.u
        } : null;
    } catch  {
        return null;
    }
}
